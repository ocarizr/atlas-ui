cmake_minimum_required(VERSION 3.25)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build: cmake --preset windows-msvc")
endif()

option(ATLAS_ENABLE_ASAN "Instrument Atlas targets with AddressSanitizer" OFF)
option(ATLAS_ENABLE_TSAN "Reserved: unsupported on native Windows" OFF)
option(ATLAS_ENABLE_COVERAGE "Instrument Atlas targets with LLVM coverage" OFF)
option(ATLAS_ENABLE_CLANG_TIDY "Analyze Atlas targets with clang-tidy" OFF)
option(ATLAS_ENABLE_MSVC_ANALYSIS "Analyze Atlas targets with MSVC /analyze" OFF)
option(ATLAS_USE_PREPARED_DEPENDENCIES "Use installed products without package installation" OFF)
include(cmake/atlas_lua_development.cmake)

set(ATLAS_VERSION "0.1.0" CACHE STRING "Atlas semantic version")
if(NOT ATLAS_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "ATLAS_VERSION must be MAJOR.MINOR.PATCH without leading zeros.")
endif()
if(NOT ATLAS_USE_PREPARED_DEPENDENCIES AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/.tools/vcpkg/scripts/buildsystems/vcpkg.cmake"
        CACHE FILEPATH "Pinned vcpkg toolchain")
endif()
if(NOT ATLAS_USE_PREPARED_DEPENDENCIES AND NOT EXISTS "${CMAKE_TOOLCHAIN_FILE}")
    message(FATAL_ERROR "Run scripts/bootstrap_dependencies.ps1 before configuring.")
endif()
project(atlas_ui VERSION ${ATLAS_VERSION} LANGUAGES CXX)
include(cmake/atlas_platform.cmake)
option(ATLAS_BUILD_WINDOWS_INTEGRATION "Build Windows SDK adapters and integration tests" ${WIN32})
if(ATLAS_BUILD_WINDOWS_INTEGRATION AND NOT WIN32)
    message(FATAL_ERROR "Windows integration requires a Windows target.")
endif()

if(MSVC)
    include(cmake/platform/msvc.cmake)
endif()

set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(cmake/atlas_instrumentation.cmake)
include(cmake/atlas_target_options.cmake)
include(CTest)
if(BUILD_TESTING)
    include(GoogleTest)
endif()
include(cmake/atlas_components.cmake)
include(cmake/atlas_exports.cmake)
include(cmake/atlas_dependencies.cmake)
if(windows IN_LIST atlas_build_components)
    add_subdirectory(libs/platform/windows)
endif()
foreach(component diagnostics jobs mvvm mvvm-content scripting input ui ui-mvvm ui-scripting window render-host renderer ui-renderer memory renderer-jobs jobs-diagnostics streaming renderer-content ui-images ipc application)
    atlas_add_library_component(${component})
endforeach()
include(cmake/atlas_applications.cmake)
if(mvvm IN_LIST atlas_requested_components)
    add_subdirectory(benchmarks/mvvm)
endif()
if(scripting IN_LIST atlas_requested_components)
    add_subdirectory(benchmarks/scripting)
endif()
if(ui IN_LIST atlas_requested_components)
    add_subdirectory(benchmarks/ui)
endif()
if(memory IN_LIST atlas_requested_components)
    add_subdirectory(benchmarks/memory)
endif()
if(jobs-benchmark IN_LIST atlas_requested_components)
    add_subdirectory(benchmarks/jobs)
endif()
if(renderer-jobs IN_LIST atlas_requested_components OR renderer-readiness IN_LIST atlas_requested_components)
    add_subdirectory(benchmarks/renderer)
endif()
if(BUILD_TESTING)
    foreach(component ui-scripting scripting ui-mvvm mvvm-content mvvm input ui-images ui window render-host renderer ui-renderer renderer-readiness memory memory-integration diagnostics jobs renderer-jobs jobs-diagnostics streaming renderer-content ipc dependencies application windows)
        if(component IN_LIST atlas_requested_components)
            if(component STREQUAL "windows")
                add_subdirectory(tests/platform/windows)
            else()
                string(REPLACE "-" "_" directory "${component}")
                add_subdirectory(tests/${directory})
            endif()
        endif()
    endforeach()
endif()
if(scripting IN_LIST atlas_requested_components AND NOT ATLAS_SHIPPING_BUILD)
    add_subdirectory(tools/lua_debug_host)
endif()
if(WIN32 AND (game IN_LIST atlas_requested_components OR editor IN_LIST atlas_requested_components
    OR tools IN_LIST atlas_requested_components))
    include(cmake/atlas_packaging.cmake)
endif()
