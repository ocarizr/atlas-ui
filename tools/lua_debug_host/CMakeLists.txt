add_executable(atlas_lua_debug_host main.cpp)
target_include_directories(atlas_lua_debug_host PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(TARGET atlas_scripting_test_development)
    target_link_libraries(atlas_lua_debug_host PRIVATE atlas_scripting_test_development)
else()
    target_link_libraries(atlas_lua_debug_host PRIVATE atlas::scripting)
endif()
if(ATLAS_TARGET_PLATFORM STREQUAL "windows" AND ATLAS_BUILD_WINDOWS_INTEGRATION)
    target_sources(atlas_lua_debug_host PRIVATE "${PROJECT_SOURCE_DIR}/libs/platform/windows/scripting/debug_transport.cpp")
    target_link_libraries(atlas_lua_debug_host PRIVATE ws2_32)
    target_compile_definitions(atlas_lua_debug_host PRIVATE ATLAS_NATIVE_LUA_DEBUG_HOST=1)
else()
    target_compile_definitions(atlas_lua_debug_host PRIVATE ATLAS_NATIVE_LUA_DEBUG_HOST=0)
endif()
atlas_configure_target(atlas_lua_debug_host)
if(BUILD_TESTING)
    add_test(NAME lua_debug_host_version COMMAND atlas_lua_debug_host --version)
    set_tests_properties(lua_debug_host_version PROPERTIES LABELS scripting TIMEOUT 30)
    if(ATLAS_BUILD_WINDOWS_INTEGRATION)
        add_subdirectory("${PROJECT_SOURCE_DIR}/tests/platform/windows/scripting" "native_tests")
    endif()
endif()
