add_library(atlas_renderer_content STATIC src/images.cpp src/loader.cpp src/gpu_content.cpp)
add_library(atlas::renderer_content ALIAS atlas_renderer_content)
target_include_directories(atlas_renderer_content PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(atlas_renderer_content PUBLIC cxx_std_20)
target_link_libraries(atlas_renderer_content PUBLIC atlas::renderer atlas::streaming)
atlas_configure_target(atlas_renderer_content)
if(ATLAS_TARGET_PLATFORM STREQUAL "windows" AND ATLAS_BUILD_WINDOWS_INTEGRATION)
    target_sources(atlas_renderer_content PRIVATE ../platform/windows/renderer_content/image_decoder.cpp)
    target_compile_definitions(atlas_renderer_content PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(atlas_renderer_content PRIVATE windowscodecs ole32)
else()
    target_sources(atlas_renderer_content PRIVATE src/unavailable_decoder.cpp)
endif()
