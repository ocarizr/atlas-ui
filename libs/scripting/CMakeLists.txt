set(atlas_scripting_sources src/session.cpp src/dispatch.cpp src/lua_runtime.cpp src/lua_bindings.cpp src/completions.cpp src/observers.cpp src/debugger_session.cpp)
add_library(atlas_scripting STATIC ${atlas_scripting_sources})
if(ATLAS_ENABLE_LUA_DEBUGGER)
    target_sources(atlas_scripting PRIVATE src/debugger.cpp src/debugger_hooks.cpp src/debugger_transport.cpp)
else()
    target_sources(atlas_scripting PRIVATE src/debugger_disabled.cpp)
endif()
target_compile_definitions(atlas_scripting PRIVATE ATLAS_LUA_DEBUGGER_ENABLED=$<BOOL:${ATLAS_ENABLE_LUA_DEBUGGER}>)
add_library(atlas::scripting ALIAS atlas_scripting)
target_compile_features(atlas_scripting PUBLIC cxx_std_20)
target_include_directories(atlas_scripting PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(atlas_scripting PUBLIC atlas::mvvm PRIVATE unofficial::lua::lua-cpp)

atlas_export_library(atlas_scripting scripting scripting)
