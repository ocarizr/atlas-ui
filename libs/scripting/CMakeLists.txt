add_library(atlas_scripting STATIC ${atlas_scripting_sources})
target_sources(atlas_scripting PRIVATE src/domain.cpp src/domain_instances.cpp src/domain_dispatch.cpp src/lua_listeners.cpp src/listener_sync.cpp)

    target_sources(atlas_scripting PRIVATE src/debugger.cpp src/debugger_hooks.cpp src/debugger_transport.cpp)
