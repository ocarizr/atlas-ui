add_library(atlas_localization STATIC src/catalog.cpp src/context.cpp src/publication.cpp src/xml.cpp)
add_library(atlas::localization ALIAS atlas_localization)
target_include_directories(atlas_localization PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(atlas_localization PUBLIC cxx_std_20)
target_link_libraries(atlas_localization PUBLIC atlas::diagnostics PRIVATE pugixml::pugixml)
atlas_configure_target(atlas_localization)
if(ATLAS_TARGET_PLATFORM STREQUAL "windows" AND ATLAS_BUILD_WINDOWS_INTEGRATION)
    target_sources(atlas_localization PRIVATE ../platform/windows/localization/plural.cpp)
    target_compile_definitions(atlas_localization PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(atlas_localization PRIVATE icu)
else()
    target_sources(atlas_localization PRIVATE src/unavailable_plural.cpp)
endif()
atlas_export_library(atlas_localization localization localization)
